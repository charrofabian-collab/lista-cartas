  <script>
    const TOTAL = 630;
    const STORAGE_KEY = 'coleccion_1_630_estado';
    const grid = document.getElementById('numberGrid');
    const searchInput = document.getElementById('searchInput');
    const filterSelect = document.getElementById('filterSelect');

    const totalCount = document.getElementById('totalCount');
    const haveCount = document.getElementById('haveCount');
    const notHaveCount = document.getElementById('notHaveCount');
    const pendingCount = document.getElementById('pendingCount');

    function loadState() {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) return JSON.parse(saved);
      const initial = {};
      for (let i = 1; i <= TOTAL; i++) initial[i] = 'pending';
      return initial;
    }

    let state = loadState();

    function saveState() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    }

    function nextState(current) {
      if (current === 'pending') return 'have';
      if (current === 'have') return 'nothave';
      return 'pending';
    }

    function labelFor(status) {
      if (status === 'have') return 'La tengo';
      if (status === 'nothave') return 'No la tengo';
      return 'Sin marcar';
    }

    function matchesFilter(num, status) {
      const search = searchInput.value.trim();
      const filter = filterSelect.value;

      const matchesSearch = !search || String(num).includes(search);
      const matchesState = filter === 'all' || filter === status;

      return matchesSearch && matchesState;
    }

    function render() {
      grid.innerHTML = '';
      let have = 0, nothave = 0, pending = 0;

      for (let i = 1; i <= TOTAL; i++) {
        const status = state[i] || 'pending';
        if (status === 'have') have++;
        else if (status === 'nothave') nothave++;
        else pending++;

        if (!matchesFilter(i, status)) continue;

        const item = document.createElement('div');
        item.className = `item ${status}`;
        item.innerHTML = `
          <div class="num">${i}</div>
          <div class="state">${labelFor(status)}</div>
        `;

        item.addEventListener('click', () => {
          state[i] = nextState(status);
          saveState();
          render();
        });

        grid.appendChild(item);
      }

      totalCount.textContent = TOTAL;
      haveCount.textContent = have;
      notHaveCount.textContent = nothave;
      pendingCount.textContent = pending;
    }

    searchInput.addEventListener('input', render);
    filterSelect.addEventListener('change', render);

    document.getElementById('markAllHave').addEventListener('click', () => {
      for (let i = 1; i <= TOTAL; i++) {
        const status = state[i] || 'pending';
        if (matchesFilter(i, status)) state[i] = 'have';
      }
      saveState();
      render();
    });

    document.getElementById('markAllNotHave').addEventListener('click', () => {
      for (let i = 1; i <= TOTAL; i++) {
        const status = state[i] || 'pending';
        if (matchesFilter(i, status)) state[i] = 'nothave';
      }
      saveState();
      render();
    });

    document.getElementById('clearAll').addEventListener('click', () => {
      for (let i = 1; i <= TOTAL; i++) state[i] = 'pending';
      saveState();
      render();
    });

    render();
  </script>
